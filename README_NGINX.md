# NGINX CONFIG
These are the nginx configs for ojs.zemaljskimuzej.ba, deployed to Laravel Forge.

```
# FORGE CONFIG (DO NOT REMOVE!)
include forge-conf/ojs.zemaljskimuzej.ba/before/*;

server {
listen 80;
listen [::]:80;
server_name ojs.zemaljskimuzej.ba;
return 301 https://$server_name$request_uri;
}

server {
http2 on;
listen 443 ssl;
listen [::]:443 ssl;
server_name ojs.zemaljskimuzej.ba;
server_tokens off;
root /home/forge/ojs.zemaljskimuzej.ba/ojs;

    # FORGE SSL (DO NOT REMOVE!)
    ssl_certificate /etc/nginx/ssl/ojs.zemaljskimuzej.ba/2763402/server.crt;
    ssl_certificate_key /etc/nginx/ssl/ojs.zemaljskimuzej.ba/2763402/server.key;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    ssl_dhparam /etc/nginx/dhparams.pem;

    add_header X-Frame-Options "SAMEORIGIN";
    add_header X-XSS-Protection "1; mode=block";
    add_header X-Content-Type-Options "nosniff";

    index index.html index.htm index.php;

    charset utf-8;

    # FORGE CONFIG (DO NOT REMOVE!)
    include forge-conf/ojs.zemaljskimuzej.ba/server/*;

location / {
try_files $uri $uri/ @ojs;
}

location @ojs {
rewrite ^/(.*)$ /index.php/$1 last;
}

    # Handle static JS files only - exclude dynamic routes through index.php
    # The negative lookahead (?!.*/index\.php/) prevents matching URLs like
    # /index.php/glasnik/api/v1/_i18n/ui.js which are dynamically generated by PHP
    location ~* ^(?!.*/index\.php/).*\.(js|mjs)$ {
        default_type text/javascript;
        expires 1y;
        add_header Cache-Control "public, immutable";
        try_files $uri =404;
    }


    location = /favicon.ico { access_log off; log_not_found off; }
    location = /robots.txt  { access_log off; log_not_found off; }

    access_log off;
    error_log  /var/log/nginx/ojs.zemaljskimuzej.ba-error.log error;

    error_page 404 /index.php;

location ~ ^/index\.php(/|$) {
fastcgi_split_path_info ^(.+\.php)(/.*)$;
fastcgi_pass unix:/var/run/php/php8.3-fpm.sock;
include fastcgi_params;
fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
fastcgi_param PATH_INFO $fastcgi_path_info if_not_empty;
}


    location ~ /\.(?!well-known).* {
        deny all;
    }
}

# FORGE CONFIG (DO NOT REMOVE!)
include forge-conf/ojs.zemaljskimuzej.ba/after/*;
```
